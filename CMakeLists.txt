cmake_minimum_required(VERSION 3.22)
project(HoltertDiPolter VERSION 0.1.0 LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(HoltertDiPolter COMPANY_NAME "HoltertDiPolter" BUNDLE_ID "com.holtertdipolter.rumblemaker" PLUGIN_MANUFACTURER_CODE HdPt PLUGIN_CODE HdR1 FORMATS VST3 PRODUCT_NAME "HoltertDiPolter Rumble Maker" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(HoltertDiPolter)
target_sources(HoltertDiPolter PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_features(HoltertDiPolter PRIVATE cxx_std_17)
target_compile_definitions(HoltertDiPolter PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(HoltertDiPolter PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
